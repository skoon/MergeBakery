```ts
// src/ui/autosave.ts
export interface AutosaveOptions {
  /** Delay after the last store notification before saving. */
  delayMs: number;
  /** Calls flush when the page is hidden or unloaded; returns an unsubscribe function. */
  onHide: (flush: () => void) => () => void;
}
/**
 * Subscribes to the store. Each notification (re)starts a delayMs timer; when it fires,
 * calls save(store.getState()). onHide's flush saves at once if a save is pending (and
 * cancels the timer). Returns a stop function that unsubscribes and cancels the timer.
 */
export function startAutosave(
  store: GameStore,
  save: (state: GameState) => void,
  options: AutosaveOptions,
): () => void;
```
