Contract:

```ts
// board.ts — cells are row-major: index = row * board.cols + col
export function getCell(board: Board, index: CellIndex): Cell; // throws RangeError when out of range
export function setCell(board: Board, index: CellIndex, cell: Cell): Board; // returns a new Board; throws RangeError when out of range
export function toIndex(board: Board, col: number, row: number): CellIndex; // throws RangeError when off the board
export function toPos(
  board: Board,
  index: CellIndex,
): { col: number; row: number };
/** Orthogonal neighbors on the board, in the order up, right, down, left. */
export function neighbors(board: Board, index: CellIndex): CellIndex[];
/** Indexes of empty cells, ascending. */
export function emptyCells(board: Board): CellIndex[];
/**
 * The empty cell closest to `from` by straight-line distance between cell centers.
 * `from` itself counts if it is empty. Ties go to the lowest index. Null when no cell is empty.
 */
export function nearestEmpty(board: Board, from: CellIndex): CellIndex | null;

// testing.ts — imported only by tests
export type CellSpec =
  ItemId | { readonly itemId: ItemId; readonly cobwebbed: boolean } | LockKind;
/** loadGameData(), loaded once. */
export const testData: GameData;
/**
 * createNewGame(testData, 1, 0) with every board cell emptied, then `cells` placed.
 * A string equal to "crate" or "flourSack" is a lock; any other string is an item id.
 * Generator items get full charges from testData.generators. `overrides` replace
 * top-level GameState fields.
 */
export function stateWith(
  cells: Readonly<Record<number, CellSpec>>,
  overrides?: Partial<GameState>,
): GameState;
```

Rules: pure functions in src/core; no edits to types.ts; follow docs/briefs/working-rules.md.
