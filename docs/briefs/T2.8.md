/** An Action without `now`; the store adds it. */
export type ActionInput = Action extends infer A
  ? A extends Action
    ? Omit<A, 'now'>
    : never
  : never;
export type StoreListener = (
  state: GameState,
  events: readonly GameEvent[],
) => void;
export interface GameStore {
  readonly data: GameData;
  getState(): GameState;
  /**
   * Adds now = clock(), runs dispatch, and on ok keeps the new state. Listeners are called,
   * in subscription order, only when the state object changed or events is non-empty.
   * Returns the result so callers can react to a rejection.
   */
  dispatch(action: ActionInput): ActionResult;
  subscribe(listener: StoreListener): () => void;
}
export function createStore(
  data: GameData,
  initial: GameState,
  dispatch: Dispatch,

## Board view behavior

- Preload a texture for every item with `loadTexture` (src/render/assets.ts) before the first draw.
- Redraw from state whenever the store notifies and when the renderer resizes. A full redraw of 63 cells is fine.
- Cells: a cream tile with a thin crust-colored border. Crates: a brown box with cross planks. Flour sacks: a beige rounded sack. Draw locks and tiles with PixiJS `Graphics`; take colors from the palette in src/ui/tokens.css. Items: a sprite scaled to 86% of the cell, centered. Cobwebbed items: the sprite at 60% alpha with a few white web lines over it.
- Input uses PixiJS pointer events (mouse and touch). Give the stage `eventMode = 'static'` and `hitArea = app.screen`. Set CSS `touch-action: none` on the canvas.
- Drag: a pointer-down on a non-cobwebbed item becomes a drag once the pointer moves more than 6 px. The sprite follows the pointer at 1.15× scale above the other items; every cell where `canMerge` with the dragged item is true glows with a butter-yellow ring; `setDragActive(true)`. On pointer-up: if `dropZoneAt` finds a zone, call it with the source cell; else if `cellAt` finds a cell, `store.dispatch({ type: 'drop', from, to })`. A rejection or no target returns the sprite to its cell. Then `setDragActive(false)`.
- Tap (pointer-up without a drag): a generator item (`data.generators.has(itemId)`) dispatches `tapGenerator`; an item with a `collectReward` dispatches `collectBonus`. Ignore rejections for now.
- Canvas coordinates equal client coordinates minus the canvas's bounding rect origin (the app uses `autoDensity`).
